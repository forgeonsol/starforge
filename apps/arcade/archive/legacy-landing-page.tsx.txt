  return (
    <>
      <svg className="i-moon" viewBox="0 0 24 24" fill="currentColor" aria-hidden="true"><path d="M20.4 14.6A8.5 8.5 0 0 1 9.4 3.6a8.5 8.5 0 1 0 11 11Z" /></svg>
      <svg className="i-sun" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" aria-hidden="true"><circle cx="12" cy="12" r="4.2" /><path d="M12 2.5v2M12 19.5v2M4.6 4.6l1.4 1.4M18 18l1.4 1.4M2.5 12h2M19.5 12h2M4.6 19.4 6 18M18 6l1.4-1.4" /></svg>
    </>
  );
}

export default function Home() {
  const [lang, setLang] = useState<Lang>("en");
  const [ship, setShip] = useState(0);
  const rootRef = useRef<HTMLElement>(null);
  const t = copy[lang];

  // Restore the saved language (shared with the game) after hydration.
  useEffect(() => {
    const id = requestAnimationFrame(() => {
      try { if (localStorage.getItem("signal-language") === "zh") setLang("zh"); } catch {}
    });
    return () => cancelAnimationFrame(id);
  }, []);

  // Scroll-driven reveals and the hero "device" zoom.
  useEffect(() => {
    const root = rootRef.current;
    if (!root) return;
    const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    const nodes = root.querySelectorAll<HTMLElement>("[data-reveal]");
    if (reduce || !("IntersectionObserver" in window)) { nodes.forEach((n) => n.classList.add("in")); return; }
    const io = new IntersectionObserver((entries) => {
      for (const e of entries) if (e.isIntersecting) { e.target.classList.add("in"); io.unobserve(e.target); }
    }, { rootMargin: "0px 0px -8% 0px", threshold: 0.12 });
    nodes.forEach((n) => io.observe(n));
    let frame = 0;
    const onScroll = () => {
      cancelAnimationFrame(frame);
      frame = requestAnimationFrame(() => {
        const p = Math.min(1, Math.max(0, window.scrollY / (window.innerHeight * 0.9)));
        root.style.setProperty("--hero-p", p.toFixed(3));
      });
    };
    onScroll();
    window.addEventListener("scroll", onScroll, { passive: true });
    return () => { io.disconnect(); window.removeEventListener("scroll", onScroll); cancelAnimationFrame(frame); };
  }, []);

  // Hangar turntable.
  useEffect(() => {
    if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;
    const id = window.setInterval(() => setShip((s) => (s + 1) % shipShowcase.length), 2600);
    return () => window.clearInterval(id);
  }, []);

  const changeLanguage = (next: Lang) => {
    setLang(next);
    try { localStorage.setItem("signal-language", next); } catch {}
  };
  const toggleTheme = () => {
    const el = document.documentElement;
    const current = el.dataset.theme || (window.matchMedia("(prefers-color-scheme: dark)").matches ? "dark" : "light");
    const next = current === "dark" ? "light" : "dark";
    el.dataset.theme = next;
    try { localStorage.setItem("signal-theme", next); } catch {}
  };

  const [headA, headB] = t.headline.split("\n");
  const [hangarA, hangarB] = t.hangarTitle.split("\n");

  return (
    <main className="landing" lang={lang} ref={rootRef}>
      <header className="l-nav">
        <div className="l-nav-inner">
          <a className="wordmark" href="#top" aria-label="Signal Protocol home"><span className="mark">N/</span><span>SIGNAL PROTOCOL<small>NEURAL DRIFT</small></span></a>
          <nav aria-label={lang === "en" ? "Main navigation" : "主导航"}>
            <a href="#systems">{t.nav[0]}</a><a href="#sectors">{t.nav[1]}</a><a href="#season">{t.nav[2]}</a>
          </nav>
          <div className="l-actions">
            <button type="button" className="icon-pill" onClick={toggleTheme} aria-label={lang === "en" ? "Toggle light or dark appearance" : "切换浅色或深色外观"}><ThemeIcon /></button>
            <div className="lang-switch" role="group" aria-label="Language">
              <button type="button" className={lang === "en" ? "selected" : ""} aria-pressed={lang === "en"} onClick={() => changeLanguage("en")}>EN</button>
              <button type="button" className={lang === "zh" ? "selected" : ""} aria-pressed={lang === "zh"} onClick={() => changeLanguage("zh")}>中文</button>
            </div>
            <a className="btn btn-sm" href="/game.html">{t.play}</a>
          </div>
        </div>
      </header>

      <section className="l-hero" id="top">
        <div className="l-hero-copy">
          <span className="chip"><i />{t.eyebrow}</span>
          <h1><span>{headA}</span>{headB && <span className="grad">{headB}</span>}</h1>
          <p>{t.intro}</p>
          <div className="l-cta-row">
            <a href="/game.html" className="btn btn-lg">{t.play}</a>
            <a className="link" href="#systems">{t.scroll} <span aria-hidden="true">›</span></a>
          </div>
        </div>
        <div className="l-device" aria-hidden="true">
          <div className="l-device-screen">
            <img src="/arena-hero.webp" alt="" fetchPriority="high" />
            <div className="l-device-hud"><span><i />NEURAL DRIFT</span><span>SEASON 01</span><span>LIVE SIMULATION</span></div>
          </div>
        </div>
        <div className="l-hero-coordinate">NEURAL DRIFT <span>·</span> SEASON 01 <span>·</span> LIVE SIMULATION</div>
      </section>

      <section className="l-band" aria-label={lang === "en" ? "Game highlights" : "游戏特色"}>
        {t.band.map((item, i) => (
          <div key={item} data-reveal style={{ transitionDelay: `${i * 70}ms` }}><b>{pad(i + 1)}</b><span>{item}</span></div>
        ))}
      </section>

      <section className="l-section" id="systems">
        <div className="l-heading center" data-reveal>
          <span className="eyebrow">{t.sectionLabel}</span>
          <h2>{t.sectionTitle}</h2>
          <p>{t.sectionIntro}</p>
        </div>
        <div className="l-stages">
          {t.stages.map(([tag, title, body], i) => (
            <article className={`l-card stage stage-${i + 1}`} key={tag} data-reveal style={{ transitionDelay: `${i * 90}ms` }}>
              <div className="stage-art"><img src={stageArt[i]} alt="" loading="lazy" /></div>
              <div className="stage-copy">
                <span className="eyebrow">{tag}</span>
                <h3>{title}</h3>
                <p>{body}</p>
              </div>
            </article>
          ))}
        </div>
      </section>

      <section className="l-hangar surface-dark">
        <div className="l-hangar-stage" data-reveal>
          <div className="pedestal" />
          <div className="orbit o1" /><div className="orbit o2" />
          {shipShowcase.map((n, i) => (
            <img key={n} className={`hero-ship ${i === ship ? "on" : ""}`} src={`/assets/ships/ship-${pad(n)}.webp`} alt="" loading={i < 2 ? "eager" : "lazy"} />
          ))}
          <div className="tag-chip left">HULL 04 / PRISMATIC</div>
          <div className="tag-chip right">8 PROJECTILE CLASSES</div>
          <div className="ship-dots" aria-hidden="true">{shipShowcase.map((n, i) => <i key={n} className={i === ship ? "on" : ""} />)}</div>
        </div>
        <div className="l-hangar-copy" data-reveal>
          <span className="eyebrow">{t.hangarTag}</span>
          <h2><span>{hangarA}</span>{hangarB && <span className="grad">{hangarB}</span>}</h2>
          <p>{t.hangarText}</p>
          <div className="meter">
            <div><span>HANGAR COLLECTION</span><b>10 <i>/</i> 40</b></div>
            <span className="meter-track"><i /></span>
          </div>
          <a className="link" href="/game.html#hangar">{t.hangarCta} <span aria-hidden="true">›</span></a>
        </div>
        <div className="l-marquee" aria-hidden="true">
          <div className="l-marquee-track">
            {[...marqueeShips, ...marqueeShips].map((n, i) => <img key={i} src={`/assets/ships/ship-${pad(n)}.webp`} alt="" loading="lazy" />)}
          </div>
        </div>
      </section>

      <section className="l-section" id="sectors">
        <div className="l-heading split" data-reveal>
          <div><span className="eyebrow">{t.worldTag}</span><h2>{t.worldTitle}</h2></div>
          <a href="/game.html#sectors" className="link">{lang === "en" ? "View unlock path" : "查看解锁路线"} <span aria-hidden="true">›</span></a>
        </div>
        <div className="l-sectors">
          {t.worlds.map(([level, name, description], i) => (
            <article className={`l-sector sector-${i + 1}`} key={level} data-reveal style={{ transitionDelay: `${i * 80}ms` }}>
              <img src={`/assets/maps/${sectorArt[i]}.webp`} alt="" loading="lazy" />
              <span className="sector-number">{level}</span>
              <div className="sector-copy"><small>SECTOR {level.padStart(2, "0")}</small><h3>{name}</h3><p>{description}</p></div>
            </article>
          ))}
        </div>
      </section>

      <section className="l-season surface-dark" id="season">
        <div className="l-season-art" data-reveal>
          <div className="ring r1" /><div className="ring r2" /><div className="ring r3" />
          <img src="/boss-manta-v13.webp" alt="" loading="lazy" />
          <div className="season-score">01 <span>/ DAILY BOARD</span></div>
        </div>
        <div className="l-season-copy" data-reveal>
          <span className="eyebrow">{t.seasonTag}</span>
          <h2>{t.seasonTitle}</h2>
          <p>{t.seasonText}</p>
          <a href="/game.html#leaderboard" className="btn">{t.seasonCta}</a>
        </div>
      </section>

      <section className="l-controls">
        <div data-reveal><span className="eyebrow">{t.controlsTag}</span><h2>{t.controlsTitle}</h2></div>
        <div className="keys" data-reveal>
          {t.controls.map(([key, label]) => <div key={key}><kbd>{key}</kbd><span>{label}</span></div>)}
        </div>
      </section>

      <section className="l-final" data-reveal>
        <h2 className="grad">{t.finalTitle}</h2>
        <p>{t.finalText}</p>
        <a href="/game.html" className="btn btn-lg">{t.play}</a>
      </section>

      <footer className="l-footer">
        <div className="l-footer-row">
          <a className="wordmark" href="#top"><span className="mark">N/</span><span>SIGNAL PROTOCOL<small>NEURAL DRIFT</small></span></a>
          <p>{t.footer}</p>
          <a className="link" href="/game.html">{t.play} <span aria-hidden="true">›</span></a>
        </div>
        <small>© 2026 SIGNAL PROTOCOL · SEASON REWARDS ARE IN-GAME ITEMS</small>
      </footer>
    </main>
  );
}
